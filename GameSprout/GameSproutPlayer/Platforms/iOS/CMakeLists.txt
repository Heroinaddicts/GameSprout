cmake_minimum_required(VERSION 3.28)
project(GameSproutIOSPlayer LANGUAGES C CXX OBJC OBJCXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    message(FATAL_ERROR "Configure GameSproutIOSPlayer with -DCMAKE_SYSTEM_NAME=iOS.")
endif()
if(NOT IS_DIRECTORY "${GAMESPROUT_IOS_PAYLOAD}/Artifact" OR
   NOT EXISTS "${GAMESPROUT_IOS_PAYLOAD}/Player.json")
    message(FATAL_ERROR "The prepared Player payload is missing.")
endif()
if(NOT EXISTS "${GAMESPROUT_SDL_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "The SDL source with GRHI's GPU adapter extension is required.")
endif()

get_filename_component(gamesprout_directory "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CMAKE_XCODE_GENERATE_TOP_LEVEL_PROJECT_ONLY ON)
set(CMAKE_XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2")
set(CMAKE_XCODE_ATTRIBUTE_ONLY_ACTIVE_ARCH NO)
set(GAMESPROUT_GRHI_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/SDK" CACHE PATH "" FORCE)
set(GAMESPROUT_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/SDK" CACHE PATH "" FORCE)
set(GAMESPROUT_GRHI_STATIC ON CACHE BOOL "" FORCE)
set(GAMESPROUT_RUNTIME_STATIC ON CACHE BOOL "" FORCE)
set(GAMESPROUT_GRHI_SDL_TARGET SDL3::SDL3-static CACHE STRING "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
set(SDL_VULKAN OFF CACHE BOOL "" FORCE)
set(SDL_METAL ON CACHE BOOL "" FORCE)

add_subdirectory("${GAMESPROUT_SDL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/SDL" EXCLUDE_FROM_ALL)
add_subdirectory("${gamesprout_directory}/GRHI" "${CMAKE_CURRENT_BINARY_DIR}/GRHI")
add_subdirectory("${gamesprout_directory}/GameSproutRuntime" "${CMAKE_CURRENT_BINARY_DIR}/Runtime")
set(GAMESPROUT_RUNTIME_SDK_DIR "${gamesprout_directory}/GameSproutRuntime/Api")
if(GAMESPROUT_IOS_GAMELOGIC_SOURCES)
    add_subdirectory("${GAMESPROUT_IOS_GAMELOGIC_SOURCES}" "${CMAKE_CURRENT_BINARY_DIR}/GameLogic")
endif()
add_subdirectory("${gamesprout_directory}/GameSproutPlayer" "${CMAKE_CURRENT_BINARY_DIR}/Player")
