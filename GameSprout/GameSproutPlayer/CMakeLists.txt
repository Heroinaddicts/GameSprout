cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(GameSproutPlayer LANGUAGES CXX)
endif()

include("${CMAKE_CURRENT_LIST_DIR}/../PreparedResources/PreparedResources.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../GameSproutRuntime/GameSproutRuntime.cmake")
set(player_sdl SDL3::SDL3-shared)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(player_sdl SDL3::SDL3-static)
endif()
if(WIN32)
    include("${CMAKE_CURRENT_LIST_DIR}/../Tools/FindGameSproutSDL3.cmake")
    GameSproutFindSDL3("${player_sdl}")
elseif(NOT TARGET ${player_sdl})
    find_package(SDL3 3.2 CONFIG REQUIRED HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../../../SDL/build")
endif()
set(player_sources PlayerMain.cpp PlayerPackage.cpp PlayerPackage.h PlayerInput.cpp PlayerInput.h
    PlatformInput/SdlInputDevices.cpp PlatformInput/SdlInputDevices.h)
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    add_library(GameSproutPlayer SHARED ${player_sources})
    set_target_properties(GameSproutPlayer PROPERTIES OUTPUT_NAME main)
    if(ANDROID)
        target_sources(GameSproutPlayer PRIVATE Platforms/Android/MobilePaths.cpp)
    else()
        target_sources(GameSproutPlayer PRIVATE Platforms/HarmonyOS/MobilePaths.cpp)
        target_compile_definitions(GameSproutPlayer PRIVATE GAMESPROUT_PLAYER_HARMONYOS=1)
    endif()
else()
    add_executable(GameSproutPlayer MACOSX_BUNDLE ${player_sources})
endif()
target_link_libraries(GameSproutPlayer PRIVATE ObjectDocument::ObjectDocument
    PreparedResources::PreparedResources GameSproutRuntime::GameSproutRuntimeApi ${player_sdl})
if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    set_target_properties(GameSproutPlayer PROPERTIES
        MACOSX_BUNDLE_GUI_IDENTIFIER "com.gamesprout.player"
        MACOSX_BUNDLE_BUNDLE_NAME "GameSprout Player"
        BUILD_WITH_INSTALL_RPATH TRUE
        INSTALL_RPATH "@executable_path"
    )
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set_target_properties(GameSproutPlayer PROPERTIES
        BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
endif()
if(GAMESPROUT_RUNTIME_STATIC)
    target_link_libraries(GameSproutPlayer PRIVATE GameSproutRuntime::GameSproutRuntime)
    target_compile_definitions(GameSproutPlayer PRIVATE GAMESPROUT_PLAYER_STATIC_RUNTIME=1)
else()
    GameSproutRuntimeDeploy(GameSproutPlayer)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    include("${CMAKE_CURRENT_LIST_DIR}/Platforms/iOS/PlayerIOS.cmake")
endif()
